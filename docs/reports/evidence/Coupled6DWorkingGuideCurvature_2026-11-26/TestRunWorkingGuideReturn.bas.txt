Public Function TestRunWorkingGuideReturn() As String
    ' -------------------- ДЛЯ ТЕСТОВ --------------------
    ' Проверяет штатный возврат с секущей и всеми рабочими ограничителями; второй случай исчерпывает общий бюджет до возврата.
    If mReverseFieldsTest.Count <> 25 Then Err.Raise 5, , "Нужна точная история25."
    mReverseSolverTest.DiscardStep
    Dim revision As Long: revision = mReverseSolverTest.AcceptedRevision
    Dim inputHandle As Integer: inputHandle = FreeFile
    Open mPolishRootTest & "\Input\RecordedYieldSeed.bin" For Binary Access Read As #inputHandle
    Dim meta As Variant, guide() As Double, accepted() As Double, expected() As Double, projection() As Double
    meta = TestReadPolishVector(inputHandle): guide = TestReadPolishVector(inputHandle)
    accepted = TestReadPolishVector(inputHandle): expected = TestReadPolishVector(inputHandle): projection = TestReadPolishVector(inputHandle)
    If Seek(inputHandle) <> LOF(inputHandle) + 1 Or UBound(meta) <> 10 Then Err.Raise 5, , "Неверный вход штатного возврата."
    Close #inputHandle
    Open mPolishRootTest & "\Input\GuideReturnEffort.bin" For Binary Access Read As #inputHandle
    Dim cost As Variant: cost = TestReadPolishVector(inputHandle)
    If Seek(inputHandle) <> LOF(inputHandle) + 1 Or UBound(cost) <> 3 Then Err.Raise 5, , "Неверные затраты материального корня."
    Close #inputHandle
    If meta(6) <> mReverseOptionsTest(3) Or meta(7) <> mReverseOptionsTest(4) Or meta(8) <> mReverseOptionsTest(2) Then Err.Raise 5, , "Нельзя менять допуски или исходный бюджет."
    Dim check() As Double, i As Long: mReverseSolverTest.CopyAccepted check
    For i = 1 To UBound(accepted)
        If check(i) <> accepted(i) Then Err.Raise 5, , "Изменено исходное принятое поле."
    Next i
    Dim handle As Integer: handle = FreeFile
    Open mPolishRootTest & "\Native\WorkingGuideReturn.bin" For Binary Access Write As #handle
    Dim magic As String * 16: magic = "NDMGUIDERET00001"
    Put #handle, , magic: TestWriteReverseBinaryVector handle, Array(2#)
    Dim index As Long, zero() As Double, values() As Double, matrix() As Double, equations As ICoupledEquationSystem
    Dim raw As Double, fraction As Double, reference As Double, factor As Double, solved As Boolean, details As String, steps As Long
    Dim field() As Double, actual() As Double, forces() As Double, result As CResultMeta
    mReverseTraceTest.ReadPoint mReverseTraceTest.Count, fraction, reference
    For index = 1 To 2
        ReDim zero(1 To UBound(guide) + 1)
        If Not mReverseSolverTest.BeginNewtonLineTest(mReverseBaseTest, mReverseDirectionTest, projection, CDbl(meta(5)), guide, CDbl(meta(4))) Then Err.Raise 5, , "Не открыт контроль исходного материального корня."
        Set equations = mReverseSolverTest
        If Not equations.TryEvaluate(zero, False, values, matrix) Then Err.Raise 5, , "Не прочитана исходная невязка корня."
        For i = 1 To UBound(guide)
            raw = values(i): If i > mReverseSolverTest.PhysicalDegreeCount Then raw = raw / CDbl(meta(9))
            If raw <> expected(i) Then Err.Raise 5, , "Материальный корень не воспроизведён побитно."
        Next i
        mReverseSolverTest.EndNewtonLineTest
        steps = CLng(cost(1)): If index = 2 Then steps = CLng(meta(8))
        details = vbNullString
        solved = mReverseSolverTest.TestMaterialGuideOriginalReturn(mReverseBaseTest, mReverseDirectionTest, guide, CDbl(meta(4)), _
            projection, CDbl(meta(5)), reference, CDbl(meta(6)), CDbl(meta(7)), CLng(meta(8)), steps, CLng(cost(2)), CLng(cost(3)), factor, details)
        Set result = mReverseSolverTest.ResultMeta
        TestWriteReverseBinaryVector handle, Array(index, Abs(CLng(solved)), factor, steps, cost(2), cost(3), meta(8), result.InternalStatus, result.ResultCode, revision)
        TestWriteReverseBinaryVector handle, mReverseSolverTest.ReadPolishEffortTest
        Dim logHandle As Integer: logHandle = FreeFile
        Open mPolishRootTest & "\Native\WorkingGuideReturn.Case" & CStr(index) & ".txt" For Output As #logHandle
        Print #logHandle, details: Print #logHandle, result.ResultComment: Print #logHandle, result.DiagnosticDetails
        Close #logHandle
        If solved Then
            mReverseSolverTest.CopyTrial field, actual
            mReverseSolverTest.DiscardStep
            If Not mReverseSolverTest.BeginNewtonLineTest(mReverseBaseTest, mReverseDirectionTest, projection, CDbl(meta(5)), field, factor) Then Err.Raise 5, , "Не открыт независимый контроль возвращённого поля."
            Set equations = mReverseSolverTest
            If Not equations.TryEvaluate(zero, False, values, matrix) Then Err.Raise 5, , "Не прочитано возвращённое поле."
            mReverseSolverTest.CopyEquationResultantResidual forces
            TestWriteReverseBinaryVector handle, field: TestWriteReverseBinaryVector handle, actual
            TestWriteReverseBinaryVector handle, values: TestWriteReverseBinaryVector handle, forces
            mReverseSolverTest.EndNewtonLineTest
            TestWriteRecordedYieldStressTest handle, field
        End If
        mReverseSolverTest.DiscardStep
        If mReverseSolverTest.AcceptedRevision <> revision Then Err.Raise 5, , "Штатный возврат изменил принятую историю."
    Next index
    Close #handle
    mReverseSolverTest.CopyAccepted check
    For i = 1 To UBound(accepted)
        If check(i) <> accepted(i) Then Err.Raise 5, , "Проверка изменила принятое поле."
    Next i
    TestRunWorkingGuideReturn = "TOTAL_WORKING_GUIDE_RETURN_TEST: passed=1; failed=0; numerical outcomes recorded; history unchanged"
End Function
