Public Function TestReadRecordedTailSystem() As String
    ' -------------------- ДЛЯ ТЕСТОВ --------------------
    ' Воспроизводит фактическую управляемую систему холодного отказа от точной истории25 и сравнивает два режима численного резерва.
    ' Записанные промежуточные поля не принимаются; новая точка требует отдельной независимой проверки всех сил и материала.
    If mReverseFieldsTest.Count <> 25 Then Err.Raise 5, , "Нужна полная принятая история25 холодного отказа."
    mReverseSolverTest.DiscardStep
    Dim revision As Long: revision = mReverseSolverTest.AcceptedRevision
    Dim inputHandle As Integer: inputHandle = FreeFile
    Open mPolishRootTest & "\Input\FrozenNewton.bin" For Binary Access Read As #inputHandle
    Dim meta As Variant, current() As Double, accepted() As Double, residual() As Double, rhs() As Double, movement() As Double
    meta = TestReadPolishVector(inputHandle): current = TestReadPolishVector(inputHandle)
    accepted = TestReadPolishVector(inputHandle): residual = TestReadPolishVector(inputHandle)
    rhs = TestReadPolishVector(inputHandle): movement = TestReadPolishVector(inputHandle)
    Dim size As Long: size = UBound(rhs)
    Dim original() As Double, expected() As Double, vector As Variant, row As Long, column As Long
    ReDim original(1 To size, 1 To size): ReDim expected(1 To size, 1 To size)
    For row = 1 To size
        vector = TestReadPolishVector(inputHandle)
        For column = 1 To size: original(column, row) = vector(column): Next column
    Next row
    For row = 1 To size
        vector = TestReadPolishVector(inputHandle)
        For column = 1 To size: expected(column, row) = vector(column): Next column
    Next row
    If Seek(inputHandle) <> LOF(inputHandle) + 1 Or CDbl(meta(7)) <> 0# Then Err.Raise 5, , "Неверный контракт записи управляемой системы."
    Close #inputHandle
    Dim checked() As Double: mReverseSolverTest.CopyAccepted checked
    For column = 1 To UBound(accepted)
        If checked(column) <> accepted(column) Then Err.Raise 5, , "Принятое поле не соответствует фактическому отказу."
    Next column
    Dim projection() As Double: ReDim projection(1 To UBound(current))
    For column = 1 To UBound(current): projection(column) = original(column, size) / CDbl(meta(5)): Next column
    Dim zero() As Double: ReDim zero(1 To size)
    If Not mReverseSolverTest.BeginNewtonLineTest(mReverseBaseTest, mReverseDirectionTest, projection, _
            CDbl(meta(6)), current, CDbl(meta(3))) Then Err.Raise 5, , "Не открыт точный контекст последнего отказа."
    Dim values() As Double, jacobian() As Double, scales As Variant
    scales = mReverseSolverTest.ReadNewtonLineScalesTest
    If Not mReverseSolverTest.TryReadBranchLinearization(zero, values, jacobian) Then Err.Raise 5, , "Не прочитана уточнённая касательная исходной точки."
    Dim handle As Integer: handle = FreeFile
    Open mPolishRootTest & "\Native\RecordedTailSystem.bin" For Binary Access Write As #handle
    Dim magic As String * 16: magic = "NDMTAILSYSTEM001"
    Put #handle, , magic
    TestWriteReverseBinaryVector handle, scales
    TestWriteReverseBinaryVector handle, values
    Dim outputRow() As Double, value As Double, rowMaximum As Double: ReDim outputRow(1 To size)
    For row = 1 To size - 1
        value = values(row)
        If row >= CLng(meta(4)) Then value = value / CDbl(meta(5))
        If Abs(value - residual(row)) > 0.00000000001 Then Err.Raise 5, , "Не воспроизведена невязка записанного отказа; новые пробы запрещены."
    Next row
    If Abs(values(size) + rhs(size)) > 0.00000000001 Then Err.Raise 5, , "Не воспроизведена исходная управляющая строка."
    For row = 1 To size
        rowMaximum = 0#
        For column = 1 To size: rowMaximum = GeomMax(rowMaximum, Abs(expected(column, row))): Next column
        For column = 1 To size
            outputRow(column) = jacobian(column, row) / CDbl(scales(column))
            If row >= CLng(meta(4)) And row < size Then outputRow(column) = outputRow(column) / CDbl(meta(5))
            If Abs(outputRow(column) - expected(column, row)) > 0.000000000002 * rowMaximum + 0.000000000001 Then _
                Err.Raise 5, , "Не воспроизведена уточнённая матрица; новые пробы запрещены."
        Next column
        TestWriteReverseBinaryVector handle, outputRow
    Next row
    mReverseSolverTest.EndNewtonLineTest
    If mReverseSolverTest.AcceptedRevision <> revision Then Err.Raise 5, , "Чтение матрицы изменило принятую историю."
    Dim index As Long, reference As Double, fraction As Double, acceptedLambda As Double, trialLambda As Double, solved As Boolean
    mReverseTraceTest.ReadPoint mReverseTraceTest.Count, fraction, acceptedLambda
    Dim seed As Variant, field() As Double, actual() As Double, result As CResultMeta
    Dim point As Long, side As Long, code As ESolverFailureCode, response As ICoupledPointResponse
    Dim strain(1 To 6) As Double, slope(1 To 6) As Double, stress(1 To 6) As Double, tangent(1 To 6, 1 To 6) As Double
    For index = 1 To 2
        reference = acceptedLambda
        If index = 2 Then reference = 0# ' Сравнивается весь режим резерва; нулевой reference меняет не только запрет уменьшения lambda.
        mHCorrectionCaseTest = index: mFullMeritCaseTest = index
        seed = current: trialLambda = CDbl(meta(3))
        solved = mReverseSolverTest.TryDeformationStep(mReverseBaseTest, mReverseDirectionTest, projection, CDbl(meta(6)), _
            CDbl(meta(3)), trialLambda, CDbl(mReverseOptionsTest(3)), CDbl(mReverseOptionsTest(4)), CLng(mReverseOptionsTest(2)), _
            consistentControl:=True, initialField:=seed, allowSmallCorrection:=True, allowHybridRecovery:=True, _
            conditionHybridRecovery:=True, referenceLambda:=reference, allowRefinedRestart:=True)
        Set result = mReverseSolverTest.ResultMeta
        TestWriteReverseBinaryVector handle, Array(index, reference, Abs(CLng(solved)), trialLambda, result.InternalStatus, result.ResultCode)
        TestWriteReverseBinaryVector handle, mReverseSolverTest.ReadPolishEffortTest
        Dim logHandle As Integer: logHandle = FreeFile
        Open mPolishRootTest & "\Native\RecordedTailSystem.Case" & CStr(index) & ".txt" For Output As #logHandle
        Print #logHandle, result.ResultComment
        Print #logHandle, result.DiagnosticDetails
        Close #logHandle
        If solved Then
            mReverseSolverTest.CopyTrial field, actual
            TestWriteReverseBinaryVector handle, field: TestWriteReverseBinaryVector handle, actual
            For side = -1 To 1
                Select Case side
                    Case -1: Set response = mReverseSidesTest(1)
                    Case 0: Set response = mReverseConcreteTest
                    Case 1: Set response = mReverseSidesTest(3)
                End Select
                For point = 1 To mReverseOperatorsTest.PointCount
                    mReverseOperatorsTest.PointStrain point, field, strain
                    mReverseOperatorsTest.AxialGradientAtPoint point, field, slope
                    For column = 1 To 6: strain(column) = strain(column) + side * 0.001 * slope(column): Next column
                    code = response.EvaluatePoint(point, strain, stress, tangent, ctmStressOnly, recordTrial:=False)
                    TestWriteReverseBinaryVector handle, Array(point, side, code)
                    TestWriteReverseBinaryVector handle, stress
                Next point
            Next side
        End If
        mReverseSolverTest.DiscardStep
        If mReverseSolverTest.AcceptedRevision <> revision Then Err.Raise 5, , "Проба резерва изменила принятую ревизию."
    Next index
    Close #handle
    mReverseSolverTest.CopyAccepted checked
    For column = 1 To UBound(accepted)
        If checked(column) <> accepted(column) Then Err.Raise 5, , "Отдельная система изменила принятое поле."
    Next column
    TestReadRecordedTailSystem = "TOTAL_RECORDED_TAIL_SYSTEM_TEST: passed=1; failed=0; outcomes recorded; no history acceptance or capacity claim"
End Function
